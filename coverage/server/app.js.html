/**
 * 🚀 Sensor Game Hub v6.0 Application Entry Point
 *
 * 새로운 모듈형 아키텍처의 메인 진입점
 * - GameServer 클래스 인스턴스 생성 및 실행
 * - 환경 설정 및 에러 핸들링
 * - 우아한 종료 처리
 */
 
const GameServer = require('./core/GameServer');
const path = require('path');
require('dotenv').config();
 
// 환경 설정
const NODE_ENV = process.env.NODE_ENV || 'development';
const PORT = process.env.PORT || 3000;
 
/**
 * 애플리케이션 시작
 */
async function startApplication() {
    try {
        console.log('🚀 Sensor Game Hub v6.0 시작 중...');
        console.log(`📍 환경: ${NODE_ENV}`);
        console.log(`📂 작업 디렉토리: ${process.cwd()}`);
 
        // GameServer 인스턴스 생성
        const gameServer = new GameServer({
            port: PORT,
            publicPath: path.join(__dirname, '../public'),
            uploadsPath: path.join(__dirname, '../uploads'),
            environment: NODE_ENV
        });
 
        // 서버 시작
        await gameServer.start();
 
        console.log(`✅ 서버가 성공적으로 시작되었습니다!`);
        console.log(`🌐 URL: http://localhost:${PORT}`);
        console.log(`📱 센서 클라이언트: http://localhost:${PORT}/sensor.html`);
        console.log(`🤖 AI 어시스턴트: http://localhost:${PORT}/ai/assistant`);
        console.log(`🎮 게임 허브: http://localhost:${PORT}/games`);
 
        // 서버 인스턴스를 전역에 저장 (종료 처리용)
        global.gameServer = gameServer;
 
    } catch (error) {
        console.error('❌ 서버 시작 실패:', error);
 
        // 상세한 에러 정보 출력
        if (error.code === 'EADDRINUSE') {
            console.error(`💥 포트 ${PORT}가 이미 사용 중입니다.`);
            console.error('🔧 다른 포트를 사용하려면 PORT 환경 변수를 설정하세요.');
        } else if (error.code === 'EACCES') {
            console.error(`💥 포트 ${PORT}에 대한 권한이 없습니다.`);
            console.error('🔧 sudo로 실행하거나 1024 이상의 포트를 사용하세요.');
        } else {
            console.error('💥 예상치 못한 오류:', error.message);
            if (NODE_ENV === 'development') {
                console.error('🔍 스택 트레이스:', error.stack);
            }
        }
 
        process.exit(1);
    }
}
 
/**
 * 우아한 서버 종료 처리
 */
function setupGracefulShutdown() {
    const shutdownSignals = ['SIGTERM', 'SIGINT'];
 
    shutdownSignals.forEach(signal => {
        process.on(signal, async () => {
            console.log(`\n📡 ${signal} 신호를 받았습니다. 서버를 종료합니다...`);
 
            try {
                if (global.gameServer) {
                    console.log('🔄 활성 세션 정리 중...');
                    await global.gameServer.shutdown();
                    console.log('✅ 서버가 정상적으로 종료되었습니다.');
                }
            } catch (error) {
                console.error('❌ 서버 종료 중 오류 발생:', error);
            } finally {
                process.exit(0);
            }
        });
    });
}
 
/**
 * 처리되지 않은 예외 처리
 */
function setupErrorHandlers() {
    // 처리되지 않은 Promise 거부
    process.on('unhandledRejection', (reason, promise) => {
        console.error('❌ 처리되지 않은 Promise 거부:', promise);
        console.error('💥 이유:', reason);
 
        if (NODE_ENV === 'production') {
            // 프로덕션에서는 서버를 종료
            console.error('🔄 서버를 재시작합니다...');
            process.exit(1);
        }
    });
 
    // 처리되지 않은 예외
    process.on('uncaughtException', (error) => {
        console.error('❌ 처리되지 않은 예외:', error);
        console.error('🔍 스택 트레이스:', error.stack);
 
        // 심각한 오류이므로 즉시 종료
        console.error('💥 치명적 오류로 인해 서버를 종료합니다.');
        process.exit(1);
    });
}
 
/**
 * 개발 환경 전용 기능
 */
function setupDevelopmentFeatures() {
    if (NODE_ENV === 'development') {
        // 개발 중 파일 변경 감지 (nodemon 대신 사용 가능)
        console.log('🔧 개발 모드가 활성화되었습니다.');
        console.log('🔄 파일 변경 시 수동으로 서버를 재시작하세요.');
 
        // 메모리 사용량 모니터링
        setInterval(() => {
            const memUsage = process.memoryUsage();
            const mbUsage = {
                rss: Math.round(memUsage.rss / 1024 / 1024),
                heapTotal: Math.round(memUsage.heapTotal / 1024 / 1024),
                heapUsed: Math.round(memUsage.heapUsed / 1024 / 1024),
                external: Math.round(memUsage.external / 1024 / 1024)
            };
 
            // 메모리 사용량이 높을 때만 출력
            if (mbUsage.heapUsed > 100) {
                console.log('📊 메모리 사용량:', mbUsage);
            }
        }, 30000); // 30초마다 체크
    }
}
 
/**
 * 시작 시 환경 검증
 */
function validateEnvironment() {
    const requiredVars = [
        // 'SUPABASE_URL',
        // 'SUPABASE_ANON_KEY'
    ];
 
    const missingVars = requiredVars.filter(varName => !process.env[varName]);
 
    if (missingVars.length > 0) {
        console.warn('⚠️  환경 변수 누락:', missingVars.join(', '));
        console.warn('🔧 .env 파일을 확인하거나 환경 변수를 설정하세요.');
 
        if (NODE_ENV === 'production') {
            console.error('💥 프로덕션 환경에서는 필수 환경 변수가 필요합니다.');
            process.exit(1);
        }
    }
}
 
/**
 * 애플리케이션 초기화 및 실행
 */
async function main() {
    console.log('🎮 Sensor Game Hub v6.0 - 새로운 모듈형 아키텍처');
    console.log('=' .repeat(60));
 
    // 환경 검증
    validateEnvironment();
 
    // 에러 핸들러 설정
    setupErrorHandlers();
 
    // 우아한 종료 처리 설정
    setupGracefulShutdown();
 
    // 개발 환경 기능 설정
    setupDevelopmentFeatures();
 
    // 애플리케이션 시작
    await startApplication();
}
 
// 애플리케이션 실행
if (require.main === module) {
    main().catch(error => {
        console.error('❌ 애플리케이션 시작 실패:', error);
        process.exit(1);
    });
}
 
module.exports = { startApplication, setupGracefulShutdown };
