/**
 * 🔧 Environment Validator v6.0
 *
 * 환경 변수 검증 및 관리 유틸리티
 * - 필수 환경 변수 검증
 * - 타입 변환 및 기본값 설정
 * - 보안 검증
 */
 
const fs = require('fs');
const path = require('path');
 
class EnvironmentValidator {
    constructor() {
        this.requiredVars = new Set();
        this.optionalVars = new Map();
        this.validators = new Map();
        this.transformers = new Map();
 
        // 기본 검증 규칙 설정
        this.setupDefaultRules();
 
        console.log('🔧 환경 변수 검증기 초기화 완료');
    }
 
    /**
     * 기본 검증 규칙 설정
     */
    setupDefaultRules() {
        // 필수 환경 변수
        this.requiredVars.add('NODE_ENV');
 
        // 선택 환경 변수와 기본값
        this.optionalVars.set('PORT', '3000');
        this.optionalVars.set('LOG_LEVEL', 'info');
        this.optionalVars.set('DEBUG_MODE', 'false');
        this.optionalVars.set('ENABLE_CACHE', 'true');
        this.optionalVars.set('CACHE_TTL', '300');
        this.optionalVars.set('MAX_SENSORS_PER_SESSION', '10');
        this.optionalVars.set('SENSOR_DATA_RATE_LIMIT', '20');
        this.optionalVars.set('DEFAULT_GAME_SESSION_TIMEOUT', '3600000');
 
        // 타입 변환기
        this.transformers.set('PORT', (value) => parseInt(value, 10));
        this.transformers.set('DEBUG_MODE', (value) => value === 'true');
        this.transformers.set('ENABLE_CACHE', (value) => value === 'true');
        this.transformers.set('CACHE_TTL', (value) => parseInt(value, 10));
        this.transformers.set('MAX_SENSORS_PER_SESSION', (value) => parseInt(value, 10));
        this.transformers.set('SENSOR_DATA_RATE_LIMIT', (value) => parseInt(value, 10));
        this.transformers.set('DEFAULT_GAME_SESSION_TIMEOUT', (value) => parseInt(value, 10));
 
        // 검증 규칙
        this.validators.set('PORT', (value) => {
            const port = parseInt(value, 10);
            return port > 0 && port <= 65535;
        });
 
        this.validators.set('NODE_ENV', (value) => {
            const validEnvs = ['development', 'test', 'production'];
            return validEnvs.includes(value);
        });
 
        this.validators.set('LOG_LEVEL', (value) => {
            const validLevels = ['error', 'warn', 'info', 'debug'];
            return validLevels.includes(value);
        });
    }
 
    /**
     * 환경 변수 검증 및 로드
     */
    validateAndLoad() {
        const errors = [];
        const warnings = [];
        const config = {};
 
        // .env 파일 존재 확인
        const envPath = path.join(process.cwd(), '.env');
        if (fs.existsSync(envPath)) {
            console.log('📄 .env 파일 감지됨');
        } else {
            warnings.push('.env 파일이 존재하지 않습니다. 환경 변수를 직접 설정하세요.');
        }
 
        // 필수 환경 변수 검증
        for (const varName of this.requiredVars) {
            const value = process.env[varName];
 
            if (!value) {
                errors.push(`필수 환경 변수 '${varName}'이 설정되지 않았습니다.`);
                continue;
            }
 
            if (this.validators.has(varName) && !this.validators.get(varName)(value)) {
                errors.push(`환경 변수 '${varName}'의 값이 유효하지 않습니다: ${value}`);
                continue;
            }
 
            config[varName] = this.transformers.has(varName)
                ? this.transformers.get(varName)(value)
                : value;
        }
 
        // 선택적 환경 변수 처리
        for (const [varName, defaultValue] of this.optionalVars) {
            let value = process.env[varName] || defaultValue;
 
            if (this.validators.has(varName) && !this.validators.get(varName)(value)) {
                warnings.push(`환경 변수 '${varName}'의 값이 유효하지 않아 기본값을 사용합니다: ${defaultValue}`);
                value = defaultValue;
            }
 
            config[varName] = this.transformers.has(varName)
                ? this.transformers.get(varName)(value)
                : value;
        }
 
        // 보안 관련 환경 변수 확인
        this.checkSecurityVars(warnings);
 
        // 결과 처리
        if (errors.length > 0) {
            console.error('❌ 환경 변수 검증 실패:');
            errors.forEach(error => console.error(`   ${error}`));
            throw new Error('환경 변수 설정이 올바르지 않습니다.');
        }
 
        if (warnings.length > 0) {
            console.warn('⚠️ 환경 변수 경고:');
            warnings.forEach(warning => console.warn(`   ${warning}`));
        }
 
        console.log('✅ 환경 변수 검증 완료');
        return config;
    }
 
    /**
     * 보안 관련 환경 변수 확인
     */
    checkSecurityVars(warnings) {
        const securityVars = [
            'JWT_SECRET',
            'SUPABASE_SERVICE_ROLE_KEY',
            'OPENAI_API_KEY',
            'ANTHROPIC_API_KEY'
        ];
 
        for (const varName of securityVars) {
            const value = process.env[varName];
 
            if (!value) {
                warnings.push(`보안 환경 변수 '${varName}'이 설정되지 않았습니다.`);
                continue;
            }
 
            // 보안 키 길이 검증
            if (varName === 'JWT_SECRET' && value.length < 32) {
                warnings.push(`JWT_SECRET은 최소 32자 이상이어야 합니다.`);
            }
 
            // 기본값 사용 경고
            if (value.includes('your-') || value.includes('example')) {
                warnings.push(`'${varName}'이 예제 값으로 설정되어 있습니다. 실제 값으로 변경하세요.`);
            }
        }
    }
 
    /**
     * 환경별 설정 생성
     */
    createEnvironmentConfig() {
        const config = this.validateAndLoad();
 
        return {
            // 서버 설정
            server: {
                port: config.PORT,
                nodeEnv: config.NODE_ENV,
                debugMode: config.DEBUG_MODE
            },
 
            // 데이터베이스 설정
            database: {
                url: process.env.SUPABASE_URL,
                anonKey: process.env.SUPABASE_ANON_KEY,
                serviceRoleKey: process.env.SUPABASE_SERVICE_ROLE_KEY
            },
 
            // AI 서비스 설정
            ai: {
                openaiApiKey: process.env.OPENAI_API_KEY,
                openaiModel: process.env.OPENAI_MODEL || 'gpt-4-turbo-preview',
                anthropicApiKey: process.env.ANTHROPIC_API_KEY
            },
 
            // 보안 설정
            security: {
                jwtSecret: process.env.JWT_SECRET,
                corsOrigin: process.env.CORS_ORIGIN?.split(',') || ['http://localhost:3000']
            },
 
            // 게임 설정
            game: {
                sessionTimeout: config.DEFAULT_GAME_SESSION_TIMEOUT,
                maxSensorsPerSession: config.MAX_SENSORS_PER_SESSION,
                sensorDataRateLimit: config.SENSOR_DATA_RATE_LIMIT
            },
 
            // 캐싱 설정
            cache: {
                enabled: config.ENABLE_CACHE,
                ttl: config.CACHE_TTL
            },
 
            // 로깅 설정
            logging: {
                level: config.LOG_LEVEL,
                file: process.env.LOG_FILE
            },
 
            // 모니터링 설정
            monitoring: {
                enabled: process.env.ENABLE_METRICS === 'true',
                port: parseInt(process.env.METRICS_PORT || '9090', 10)
            }
        };
    }
 
    /**
     * 개발용 환경 변수 생성
     */
    generateDevelopmentEnv() {
        const envContent = `# Sensor Game Hub v6.0 개발 환경 설정
# 자동 생성됨 - ${new Date().toISOString()}
 
# 서버 설정
PORT=3000
NODE_ENV=development
DEBUG_MODE=true
 
# 로깅 설정
LOG_LEVEL=debug
 
# 게임 설정
MAX_SENSORS_PER_SESSION=10
SENSOR_DATA_RATE_LIMIT=20
DEFAULT_GAME_SESSION_TIMEOUT=3600000
 
# 캐싱 설정
ENABLE_CACHE=true
CACHE_TTL=300
 
# 모니터링 설정
ENABLE_METRICS=false
 
# 개발 도구
ENABLE_SWAGGER=true
SWAGGER_PATH=/api-docs
 
# TODO: 다음 환경 변수들을 실제 값으로 설정하세요
# SUPABASE_URL=https://your-project.supabase.co
# SUPABASE_ANON_KEY=your-supabase-anon-key
# SUPABASE_SERVICE_ROLE_KEY=your-service-role-key
# OPENAI_API_KEY=your-openai-api-key
# JWT_SECRET=your-jwt-secret-minimum-32-characters-long
`;
 
        const envPath = path.join(process.cwd(), '.env.development');
        fs.writeFileSync(envPath, envContent);
 
        console.log(`✅ 개발 환경 파일 생성: ${envPath}`);
        return envPath;
    }
}
